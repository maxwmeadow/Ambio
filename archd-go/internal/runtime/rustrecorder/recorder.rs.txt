// Ambio's run recorder, compiled into this crate for an `investigation run`
// (docs/INVESTIGATION_RUNTIMES.md in the Ambio repository). It does nothing
// unless AMBIO_EVIDENCE_DIR is set. Standard library only.
#[allow(dead_code, unused, non_snake_case, unsafe_op_in_unsafe_fn, clippy::all)]
#[doc(hidden)]
pub(crate) mod __ambio_rec {
    use std::cell::RefCell;
    use std::collections::HashMap;
    use std::fmt::Write as _;
    use std::sync::atomic::{AtomicU32, AtomicU64, AtomicU8, Ordering};
    use std::sync::Mutex;
    use std::time::Instant;

    /// Compiled against this recorder. rustc tells Cargo which environment
    /// variables a crate read, so a different recorder (an Ambio update)
    /// makes Cargo rebuild instead of reusing a crate built with the old one.
    const VERSION: Option<&str> = option_env!("AMBIO_RECORDER_VERSION");
    /// Where the compiler ran: source paths in panics are relative to it.
    const BASE: &str = "__AMBIO_BASE__";
    const CRATE: &str = "__AMBIO_CRATE__";
    const SHOW_MAX: usize = 2000;
    const PREVIEW_MAX: usize = 240;
    /// Caller → callee counts live in a fixed table of atomics: every call
    /// records one, from any thread, and a lock there is what a program
    /// would spend its time waiting on.
    const EDGE_SLOTS: usize = 8192;
    const EDGE_PROBES: usize = 32;
    const ZERO: AtomicU64 = AtomicU64::new(0);
    static EDGE_KEYS: [AtomicU64; EDGE_SLOTS] = [ZERO; EDGE_SLOTS];
    static EDGE_COUNTS: [AtomicU64; EDGE_SLOTS] = [ZERO; EDGE_SLOTS];
    static EDGES_DROPPED: AtomicU64 = AtomicU64::new(0);

    fn edge(key: u64, calls: u64) {
        // key is caller<<32|callee, never 0: ids start at 1
        let mut at = (key.wrapping_mul(0x9E37_79B9_7F4A_7C15) >> 40) as usize & (EDGE_SLOTS - 1);
        for _ in 0..EDGE_PROBES {
            let held = EDGE_KEYS[at].load(Ordering::Relaxed);
            if held == key || (held == 0 && match EDGE_KEYS[at].compare_exchange(0, key, Ordering::Relaxed, Ordering::Relaxed) {
                Ok(_) => true,
                Err(now) => now == key,
            }) {
                EDGE_COUNTS[at].fetch_add(calls, Ordering::Relaxed);
                return;
            }
            at = (at + 1) & (EDGE_SLOTS - 1);
        }
        EDGES_DROPPED.fetch_add(calls, Ordering::Relaxed);
    }

    /// One instrumented function. A `static` inside the function it describes.
    pub struct F {
        pub name: &'static str,
        pub file: &'static str,
        pub line: u32,
        pub params: &'static str,
        /// 1: returns a Result, 2: an Option. Leaving one without a recorded
        /// value means `?` (or a macro) returned early.
        pub kind: u8,
        /// Whether return values pass through the recorder at all.
        pub captures: bool,
        /// The function's number, with WATCHED and HOT in the top bits; 0 until first called.
        id: AtomicU32,
        calls: AtomicU64,
        errors: AtomicU64,
        nanos: AtomicU64,
        /// Calls to a watched function, exactly: detail is sampled once it is busy.
        seen: AtomicU32,
    }

    const WATCHED: u32 = 0x8000_0000;
    /// Called so often that only counting it is affordable (see HOT_AFTER).
    const HOT: u32 = 0x4000_0000;
    const ID: u32 = 0x3fff_ffff;
    /// A function nobody watches is timed and placed in the call graph for
    /// its first calls; past this many it is only counted. A tiny function
    /// in a loop is called a billion times, and a clock read each way would
    /// make the program a hundred times slower.
    const HOT_AFTER: u64 = 20_000;
    /// Watched calls are all recorded in detail up to here, then every 2nd,
    /// 4th, 8th...: a few hundred samples spread over the run.
    const FULL_CALLS: u32 = 100;

    fn add_calls(f: &'static F, calls: u64, nanos: u64) {
        f.nanos.fetch_add(nanos, Ordering::Relaxed);
        if f.calls.fetch_add(calls, Ordering::Relaxed) + calls >= HOT_AFTER && f.id.load(Ordering::Relaxed) & (WATCHED | HOT) == 0 {
            f.id.fetch_or(HOT, Ordering::Relaxed);
        }
    }

    impl F {
        pub const fn new(name: &'static str, file: &'static str, line: u32, params: &'static str, kind: u8, captures: bool) -> F {
            F { name, file, line, params, kind, captures, id: AtomicU32::new(0), calls: AtomicU64::new(0), errors: AtomicU64::new(0), nanos: AtomicU64::new(0), seen: AtomicU32::new(0) }
        }
    }

    // ── showing a value ─────────────────────────────────────────────────────
    //
    // `(&&W(&value)).s()` formats the value with Debug when its type has it and
    // names the type when it does not, without the caller knowing which: method
    // lookup finds `D for &W<T>` first, and only where `T: Debug`.

    pub struct W<'a, T: ?Sized>(pub &'a T);

    pub trait D {
        fn s(&self) -> String;
    }

    impl<'a, T: ?Sized + std::fmt::Debug> D for &W<'a, T> {
        fn s(&self) -> String {
            let mut out = Capped(String::new());
            let _ = write!(out, "{:?}", self.0);
            out.0
        }
    }

    pub trait N {
        fn s(&self) -> String;
    }

    impl<'a, T: ?Sized> N for W<'a, T> {
        fn s(&self) -> String {
            let name = std::any::type_name::<T>();
            let name = name.trim_start_matches('&').trim_start_matches("mut ");
            let head = name.split('<').next().unwrap_or(name);
            format!("<{}>", head.rsplit("::").next().unwrap_or(head))
        }
    }

    // `(&&W(&value)).e()`: whether a value is a `Result::Err`, for any type,
    // by the same lookup. A function fails by returning one.

    pub trait E1 {
        fn e(&self) -> bool;
    }

    impl<'a, T, E> E1 for &W<'a, Result<T, E>> {
        fn e(&self) -> bool {
            self.0.is_err()
        }
    }

    pub trait E0 {
        fn e(&self) -> bool;
    }

    impl<'a, T: ?Sized> E0 for W<'a, T> {
        fn e(&self) -> bool {
            false
        }
    }

    /// Stops a Debug implementation once it has written enough.
    struct Capped(String);

    impl std::fmt::Write for Capped {
        fn write_str(&mut self, s: &str) -> std::fmt::Result {
            if self.0.len() + s.len() > SHOW_MAX {
                let mut room = SHOW_MAX.saturating_sub(self.0.len());
                while room > 0 && !s.is_char_boundary(room) {
                    room -= 1;
                }
                self.0.push_str(&s[..room]);
                self.0.push('…');
                return Err(std::fmt::Error);
            }
            self.0.push_str(s);
            Ok(())
        }
    }

    /// The address a reference points at: what "the same object" means.
    pub fn addr<T: ?Sized>(r: &T) -> usize {
        r as *const T as *const () as usize
    }

    // ── a parsed Debug rendering ────────────────────────────────────────────

    enum V {
        Leaf(&'static str, String), // kind, text
        Seq(Vec<V>),
        Map(Vec<(String, V)>),
        Obj(String, Vec<(String, V)>),
    }

    struct P<'a> {
        s: &'a [u8],
        i: usize,
        n: usize,
    }

    impl<'a> P<'a> {
        fn ws(&mut self) {
            while self.i < self.s.len() && (self.s[self.i] as char).is_whitespace() {
                self.i += 1;
            }
        }

        fn peek(&self) -> u8 {
            if self.i < self.s.len() { self.s[self.i] } else { 0 }
        }

        fn text(&self, from: usize) -> String {
            String::from_utf8_lossy(&self.s[from..self.i.min(self.s.len())]).into_owned()
        }

        fn quoted(&mut self, quote: u8) -> String {
            let from = self.i;
            self.i += 1;
            while self.i < self.s.len() && self.s[self.i] != quote {
                if self.s[self.i] == b'\\' {
                    self.i += 1;
                }
                self.i += 1;
            }
            self.i = (self.i + 1).min(self.s.len());
            self.text(from)
        }

        fn items(&mut self, close: u8) -> Vec<V> {
            let mut out = Vec::new();
            loop {
                self.ws();
                if self.peek() == close || self.peek() == 0 {
                    self.i += 1;
                    return out;
                }
                out.push(self.value());
                self.ws();
                if self.peek() == b',' {
                    self.i += 1;
                }
            }
        }

        fn fields(&mut self) -> Vec<(String, V)> {
            let mut out = Vec::new();
            loop {
                self.ws();
                if self.peek() == b'}' || self.peek() == 0 {
                    self.i += 1;
                    return out;
                }
                let key = self.value();
                self.ws();
                if self.peek() == b':' {
                    self.i += 1;
                    let key = match key {
                        V::Leaf("string", t) => t.trim_matches('"').to_string(),
                        V::Leaf(_, t) => t,
                        _ => "?".to_string(),
                    };
                    let value = self.value();
                    out.push((key, value));
                } else {
                    out.push((format!("[{}]", out.len()), key)); // a set
                }
                self.ws();
                if self.peek() == b',' {
                    self.i += 1;
                }
            }
        }

        fn value(&mut self) -> V {
            self.ws();
            self.n += 1;
            if self.n > 4000 {
                self.i = self.s.len();
                return V::Leaf("value", "…".to_string());
            }
            match self.peek() {
                b'"' => V::Leaf("string", self.quoted(b'"')),
                b'\'' => V::Leaf("string", self.quoted(b'\'')),
                b'[' => {
                    self.i += 1;
                    V::Seq(self.items(b']'))
                }
                b'(' => {
                    self.i += 1;
                    let items = self.items(b')');
                    if items.is_empty() { V::Leaf("value", "()".to_string()) } else { V::Seq(items) }
                }
                b'{' => {
                    self.i += 1;
                    V::Map(self.fields())
                }
                _ => {
                    let from = self.i;
                    while self.i < self.s.len() && !matches!(self.s[self.i], b',' | b')' | b']' | b'}' | b'(' | b'{' | b'[' | b':' | b'"') {
                        // `::` in a path is not a key separator
                        self.i += 1;
                    }
                    let word = self.text(from).trim().to_string();
                    self.ws();
                    match self.peek() {
                        b'(' if !word.is_empty() => {
                            self.i += 1;
                            let items = self.items(b')');
                            V::Obj(word, items.into_iter().enumerate().map(|(i, v)| (i.to_string(), v)).collect())
                        }
                        b'{' if !word.is_empty() => {
                            self.i += 1;
                            V::Obj(word, self.fields())
                        }
                        _ => {
                            if word == "true" || word == "false" {
                                V::Leaf("boolean", word)
                            } else if word.parse::<f64>().is_ok() {
                                V::Leaf("number", word)
                            } else {
                                if word.is_empty() {
                                    self.i += 1; // never stand still
                                }
                                V::Leaf("value", word)
                            }
                        }
                    }
                }
            }
        }
    }

    fn parse(text: &str) -> V {
        let mut p = P { s: text.as_bytes(), i: 0, n: 0 };
        let v = p.value();
        p.ws();
        if p.i < p.s.len() { V::Leaf("value", clip(text, 120)) } else { v }
    }

    fn clip(s: &str, max: usize) -> String {
        if s.chars().count() <= max {
            return s.to_string();
        }
        let mut out: String = s.chars().take(max - 1).collect();
        out.push('…');
        out
    }

    /// path → (kind, text), depth and count bounded.
    fn flatten(v: &V, p: &str, depth: usize, out: &mut Vec<(String, (&'static str, String))>) {
        if out.len() >= 160 {
            return;
        }
        let join = |k: &str| if p.is_empty() { k.to_string() } else if k.starts_with('[') { format!("{}{}", p, k) } else { format!("{}.{}", p, k) };
        match v {
            V::Leaf(kind, text) => out.push((p.to_string(), (kind, clip(text, 120)))),
            _ if depth >= 4 => out.push((p.to_string(), ("object", "<…>".to_string()))),
            V::Seq(items) => {
                out.push((format!("{}.length", p), ("number", items.len().to_string())));
                for (i, item) in items.iter().take(8).enumerate() {
                    flatten(item, &format!("{}[{}]", p, i), depth + 1, out);
                }
            }
            V::Map(fields) if fields.is_empty() => out.push((p.to_string(), ("object", "{}".to_string()))),
            V::Map(fields) => {
                for (k, item) in fields.iter().take(24) {
                    flatten(item, &join(&clip(k, 40)), depth + 1, out);
                }
            }
            V::Obj(name, fields) if fields.is_empty() => out.push((p.to_string(), ("object", format!("<{}>", name)))),
            V::Obj(_, fields) => {
                for (k, item) in fields.iter().take(24) {
                    flatten(item, &join(k), depth + 1, out);
                }
            }
        }
    }

    // ── what a watched function did, across its calls ───────────────────────

    #[derive(Default)]
    struct Stats {
        count: u64,
        changes: u64,
        kind: String,
        first: Option<String>,
        last: String,
        min: f64,
        max: f64,
        last_num: Option<f64>,
        falling: bool,
        rising: bool,
        series: Vec<String>,
        tail: Vec<String>,
        distinct: Vec<(String, u64)>,
        overflow: bool,
    }

    impl Stats {
        fn add(&mut self, kind: &str, text: &str) {
            self.count += 1;
            if self.kind.is_empty() {
                self.kind = kind.to_string();
            } else if self.kind != kind && self.kind != "mixed" {
                self.kind = "mixed".to_string();
            }
            if self.count > 1 && text != self.last {
                self.changes += 1;
            }
            if kind == "number" {
                if let Ok(n) = text.parse::<f64>() {
                    match self.last_num {
                        Some(prev) => {
                            if n < prev { self.falling = true; }
                            if n > prev { self.rising = true; }
                            self.min = self.min.min(n);
                            self.max = self.max.max(n);
                        }
                        None => {
                            self.min = n;
                            self.max = n;
                        }
                    }
                    self.last_num = Some(n);
                }
            }
            if self.first.is_none() {
                self.first = Some(text.to_string());
            }
            self.last = text.to_string();
            if self.series.len() < 12 {
                self.series.push(text.to_string());
            }
            self.tail.push(text.to_string());
            if self.tail.len() > 3 {
                self.tail.remove(0);
            }
            if let Some(entry) = self.distinct.iter_mut().find(|e| e.0 == text) {
                entry.1 += 1;
            } else if self.distinct.len() < 24 {
                self.distinct.push((text.to_string(), 1));
            } else {
                self.overflow = true;
            }
        }

        fn json(&self, b: &mut String) {
            let numeric = self.kind == "number";
            let trend = if numeric && self.changes > 0 {
                if !self.falling { "increasing" } else if !self.rising { "decreasing" } else { "varying" }
            } else if self.changes > 0 { "varying" } else { "constant" };
            let _ = write!(b, "{{\"kind\":{},\"count\":{},\"changes\":{},\"distinct\":", q(&self.kind), self.count, self.changes);
            if self.overflow { b.push_str("\"24+\""); } else { let _ = write!(b, "{}", self.distinct.len()); }
            let _ = write!(b, ",\"first\":{},\"last\":{},\"trend\":{},\"series\":{}", q(self.first.as_deref().unwrap_or("")), q(&self.last), q(trend), strings(&self.series));
            if numeric && self.last_num.is_some() {
                let _ = write!(b, ",\"min\":{},\"max\":{}", num(self.min), num(self.max));
            }
            if self.count > 12 {
                let _ = write!(b, ",\"tail\":{}", strings(&self.tail));
            }
            if self.distinct.len() <= 8 && !self.overflow {
                let mut top: Vec<&(String, u64)> = self.distinct.iter().collect();
                top.sort_by(|x, y| y.1.cmp(&x.1));
                b.push_str(",\"top\":[");
                for (i, e) in top.iter().enumerate() {
                    if i > 0 { b.push(','); }
                    let _ = write!(b, "{{\"value\":{},\"n\":{}}}", q(&e.0), e.1);
                }
                b.push(']');
            }
            b.push('}');
        }
    }

    struct Sample {
        call: u32,
        args: String,
        ms: f64,
        returned: Option<String>,
        threw: Option<String>,
        mutated: Vec<String>,
    }

    #[derive(Default)]
    struct Watch {
        /// Calls recorded in detail; every call is counted on the function itself.
        sampled: u32,
        errors: u32,
        total_ms: f64,
        head: Vec<Sample>,
        tail: Vec<Sample>,
        novel: Vec<Sample>,
        shapes: Vec<String>,
        paths: Vec<(String, Stats)>,
        mutations: Vec<(String, u32, Vec<String>)>,
        exceptions: Vec<(String, u32)>,
        returns: Stats,
        seen: HashMap<usize, u32>,
        shared: Vec<(String, u32, u32)>,
        last_args: Option<String>,
        repeats: u32,
        repeat_example: Option<String>,
        objects: HashMap<usize, (u32, Vec<(String, String)>)>,
        drift: Vec<(String, u32, Vec<String>)>,
    }

    impl Watch {
        fn observe(&mut self, path: &str, kind: &str, text: &str) {
            if let Some(entry) = self.paths.iter_mut().find(|e| e.0 == path) {
                entry.1.add(kind, text);
            } else if self.paths.len() < 400 {
                let mut stats = Stats::default();
                stats.add(kind, text);
                self.paths.push((path.to_string(), stats));
            }
        }

        fn sample(&mut self, s: Sample) {
            let shape = match (&s.threw, &s.returned) {
                (Some(t), _) => format!("threw:{}", t.split(':').next().unwrap_or("")),
                (None, Some(r)) => clip(r.split(|c| c == '(' || c == '{' || c == ' ').next().unwrap_or(""), 40),
                _ => "none".to_string(),
            };
            let known = self.shapes.contains(&shape);
            if !known && self.shapes.len() < 8 {
                self.shapes.push(shape);
            }
            if self.head.len() < 6 {
                self.head.push(s);
            } else if !known && self.novel.len() < 6 {
                self.novel.push(s);
            } else {
                self.tail.push(s);
                if self.tail.len() > 4 {
                    self.tail.remove(0);
                }
            }
        }

        fn json(&self, f: &F, b: &mut String) {
            let _ = write!(b, "{{\"file\":{},\"symbol\":{},\"line\":{},\"calls\":{},\"sampled\":{},\"errors\":{},\"avgMs\":{}", q(f.file), q(f.name), f.line,
                f.seen.load(Ordering::Relaxed), self.sampled, self.errors, num(if self.sampled == 0 { 0.0 } else { round3(self.total_ms / self.sampled as f64) }));
            let mut all: Vec<&Sample> = self.head.iter().chain(self.novel.iter()).chain(self.tail.iter()).collect();
            all.sort_by_key(|s| s.call);
            b.push_str(",\"samples\":[");
            for (i, s) in all.iter().enumerate() {
                if i > 0 { b.push(','); }
                let _ = write!(b, "{{\"call\":{},\"args\":{},\"ms\":{}", s.call, q(&s.args), num(round3(s.ms)));
                match (&s.threw, &s.returned) {
                    (Some(t), _) => { let _ = write!(b, ",\"threw\":{}", q(t)); }
                    (None, r) => { let _ = write!(b, ",\"returned\":{}", q(r.as_deref().unwrap_or("()"))); }
                }
                if !s.mutated.is_empty() {
                    let _ = write!(b, ",\"mutated\":{}", strings(&s.mutated));
                }
                b.push('}');
            }
            b.push_str("],\"values\":{");
            for (i, (path, stats)) in self.paths.iter().enumerate() {
                if i > 0 { b.push(','); }
                let _ = write!(b, "{}:", q(path));
                stats.json(b);
            }
            b.push_str("},\"mutations\":[");
            for (i, (path, calls, examples)) in self.mutations.iter().enumerate() {
                if i > 0 { b.push(','); }
                let _ = write!(b, "{{\"path\":{},\"calls\":{},\"examples\":[{}]}}", q(path), calls, examples.join(","));
            }
            b.push_str("],\"exceptions\":[");
            for (i, (what, n)) in self.exceptions.iter().enumerate() {
                if i > 0 { b.push(','); }
                let _ = write!(b, "{{\"what\":{},\"n\":{}}}", q(what), n);
            }
            b.push_str("],\"returns\":");
            self.returns.json(b);
            b.push_str(",\"shared\":[");
            for (i, (param, calls, first)) in self.shared.iter().enumerate() {
                if i > 0 { b.push(','); }
                let _ = write!(b, "{{\"param\":{},\"calls\":{},\"firstCall\":{}}}", q(param), calls, first);
            }
            let _ = write!(b, "],\"repeats\":{},\"repeatExample\":{},\"drift\":[", self.repeats, self.repeat_example.as_deref().unwrap_or("null"));
            for (i, (path, calls, examples)) in self.drift.iter().enumerate() {
                if i > 0 { b.push(','); }
                let _ = write!(b, "{{\"path\":{},\"calls\":{},\"examples\":[{}]}}", q(path), calls, examples.join(","));
            }
            b.push_str("]}");
        }
    }

    static ENABLED: AtomicU8 = AtomicU8::new(0); // 0: not looked yet, 1: off, 2: on
    static STATE: Mutex<Option<State>> = Mutex::new(None);
    /// AMBIO_RECORDER_DEBUG: say on stderr how evidence was flushed and written.
    fn debug(what: std::fmt::Arguments) {
        if std::env::var_os("AMBIO_RECORDER_DEBUG").is_some() {
            eprintln!("[ambio] {}: {}", CRATE, what);
        }
    }
    /// What a thread has counted and not yet added to the shared totals. A
    /// program calls its hot functions from every thread at once; counting
    /// into shared memory on each call would have the cores fighting over it.
    struct Local {
        stack: Vec<u32>,
        fns: [(Option<&'static F>, u64, u64); LOCAL_SLOTS], // function, calls, nanoseconds
        edges: [(u64, u64); LOCAL_SLOTS],                   // caller<<32|callee, calls
        flushed: Option<Instant>,
    }

    const LOCAL_SLOTS: usize = 64;

    impl Local {
        fn count(&mut self, f: &'static F, id: u32, nanos: u64, calls: u64) {
            let slot = &mut self.fns[id as usize % LOCAL_SLOTS];
            match slot.0 {
                Some(held) if std::ptr::eq(held, f) => {}
                Some(held) => {
                    add_calls(held, slot.1, slot.2);
                    *slot = (Some(f), 0, 0);
                }
                None => *slot = (Some(f), 0, 0),
            }
            slot.1 += calls;
            slot.2 += nanos;
        }

        fn edge(&mut self, from: u32, to: u32) {
            let key = ((from as u64) << 32) | to as u64;
            let slot = &mut self.edges[(from as usize * 31 + to as usize) % LOCAL_SLOTS];
            if slot.0 != key {
                if slot.1 > 0 {
                    edge(slot.0, slot.1);
                }
                *slot = (key, 0);
            }
            slot.1 += 1;
        }

        /// Adds everything held here to the shared totals.
        fn spill(&mut self) {
            for slot in self.fns.iter_mut() {
                if let Some(f) = slot.0 {
                    add_calls(f, slot.1, slot.2);
                }
                *slot = (None, 0, 0);
            }
            for slot in self.edges.iter_mut() {
                if slot.1 > 0 {
                    edge(slot.0, slot.1);
                }
                *slot = (0, 0);
            }
        }
    }

    impl Drop for Local {
        fn drop(&mut self) {
            self.spill(); // the thread is ending
        }
    }

    fn spill_this_thread() {
        let spilled = LOCAL.try_with(|l| match l.try_borrow_mut() {
            Ok(mut l) => { l.spill(); true }
            Err(_) => false,
        });
        if spilled != Ok(true) {
            debug(format_args!("could not flush this thread's counts ({})", if spilled.is_err() { "thread storage is gone" } else { "busy" }));
        }
    }

    thread_local! {
        static LOCAL: RefCell<Local> = RefCell::new(Local { stack: Vec::new(), fns: [(None, 0, 0); LOCAL_SLOTS], edges: [(0, 0); LOCAL_SLOTS], flushed: None });
        static PANIC: RefCell<Option<String>> = RefCell::new(None);
        static FRAME: RefCell<Vec<Pending>> = RefCell::new(Vec::new());
    }

    /// A watched call between entry and exit.
    struct Pending {
        id: u32,
        call: u32,
        args: String,
        before: Vec<(usize, Vec<(String, (&'static str, String))>)>,
        returned: Option<String>,
        after: Vec<(usize, String)>,
    }

    fn now_ms() -> u128 {
        std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).map(|d| d.as_millis()).unwrap_or(0)
    }

    fn on() -> bool {
        match ENABLED.load(Ordering::Relaxed) {
            2 => true,
            1 => false,
            _ => init(),
        }
    }

    fn init() -> bool {
        let dir = std::env::var("AMBIO_EVIDENCE_DIR").unwrap_or_default();
        let mut state = match STATE.lock() {
            Ok(guard) => guard,
            Err(_) => return false,
        };
        if state.is_some() {
            return true;
        }
        if dir.is_empty() {
            ENABLED.store(1, Ordering::Relaxed);
            return false;
        }
        let mut watches: Vec<(String, Vec<Spec>)> = Vec::new();
        for item in objects(&std::env::var("AMBIO_WATCHES").unwrap_or_default()) {
            let path = field(&item, "absPath").or_else(|| field(&item, "file")).unwrap_or_default();
            if path.is_empty() {
                continue;
            }
            let path = canonical(&path);
            let spec = Spec {
                symbol: field(&item, "symbol").filter(|s| !s.is_empty()).unwrap_or_else(|| "*".to_string()),
                start: field(&item, "lineStart").and_then(|s| s.parse::<f64>().ok()).unwrap_or(0.0) as u32,
                end: field(&item, "lineEnd").and_then(|s| s.parse::<f64>().ok()).unwrap_or(0.0) as u32,
            };
            match watches.iter_mut().find(|w| w.0 == path) {
                Some(entry) => entry.1.push(spec),
                None => watches.push((path, vec![spec])),
            }
        }
        *state = Some(State {
            dir,
            run_id: std::env::var("AMBIO_RUN_ID").unwrap_or_default(),
            watches,
            fns: Vec::new(),
            watched: HashMap::new(),
            uncaught: Vec::new(),
            last_written: u64::MAX,
            last_flush: Instant::now(),
            started: now_ms(),
        });
        drop(state);
        ENABLED.store(2, Ordering::Relaxed);
        // The panic's message and place, for the functions it unwinds and -
        // on the main thread, where it ends the program - as the crash.
        let previous = std::panic::take_hook();
        std::panic::set_hook(Box::new(move |info| {
            let message = if let Some(s) = info.payload().downcast_ref::<&str>() {
                s.to_string()
            } else if let Some(s) = info.payload().downcast_ref::<String>() {
                s.clone()
            } else {
                "panic".to_string()
            };
            let _ = PANIC.try_with(|p| *p.borrow_mut() = Some(clip(&message, 200)));
            if std::thread::current().name() == Some("main") {
                if let Some(at) = info.location() {
                    let file = if std::path::Path::new(at.file()).is_absolute() { at.file().to_string() } else { format!("{}/{}", BASE, at.file()) };
                    let name = LOCAL.try_with(|l| l.try_borrow().ok().and_then(|l| l.stack.last().copied())).ok().flatten();
                    spill_this_thread();
                    if let Ok(mut guard) = STATE.lock() {
                        if let Some(state) = guard.as_mut() {
                            let function = name.and_then(|id| state.fns.get(id as usize - 1)).map(|f| f.name).unwrap_or("main");
                            if state.uncaught.len() < 5 {
                                state.uncaught.push(format!("{{\"what\":{},\"stack\":{}}}", q(&format!("panic: {}", clip(&message, 300))),
                                    q(&format!("    at {} ({}:{}:1)", function, file, at.line()))));
                            }
                        }
                    }
                    write();
                }
            }
            previous(info);
        }));
        // std::process::exit skips destructors but runs the C library's exit
        // handlers, the one place code can still run.
        #[cfg(any(unix, windows))]
        unsafe {
            __AMBIO_EXTERN__ "C" {
                fn atexit(callback: extern "C" fn()) -> i32;
            }
            extern "C" fn at_exit() {
                debug(format_args!("exiting"));
                spill_this_thread();
                write();
            }
            atexit(at_exit);
        }
        // Windows' std::process::exit is ExitProcess, which skips the C exit
        // handlers but still calls each module's TLS callbacks; reading the
        // callback here keeps the linker from dropping it.
        #[cfg(windows)]
        unsafe {
            std::ptr::read_volatile(&ON_PROCESS_DETACH);
        }
        debug(format_args!("recording into {}", dir_for_debug()));
        // A process that is killed keeps what was flushed in the last second.
        let _ = std::thread::Builder::new().name("ambio-recorder-flush".to_string()).spawn(|| loop {
            std::thread::sleep(std::time::Duration::from_secs(1));
            write();
        });
        true
    }

    fn dir_for_debug() -> String {
        STATE.lock().ok().and_then(|g| g.as_ref().map(|s| s.dir.clone())).unwrap_or_default()
    }

    /// Set once the process is ending on Windows: the other threads are gone,
    /// and one may have died holding the state's lock.
    static EXITING: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);

    #[cfg(windows)]
    #[used]
    __AMBIO_TLS_SECTION__
    static ON_PROCESS_DETACH: unsafe extern "system" fn(*mut u8, u32, *mut u8) = on_process_detach;

    #[cfg(windows)]
    unsafe extern "system" fn on_process_detach(_module: *mut u8, reason: u32, _reserved: *mut u8) {
        const DLL_PROCESS_DETACH: u32 = 0;
        if reason != DLL_PROCESS_DETACH || ENABLED.load(Ordering::Relaxed) != 2 {
            return;
        }
        EXITING.store(true, Ordering::Relaxed);
        debug(format_args!("exiting"));
        spill_this_thread();
        write();
    }

    fn canonical(path: &str) -> String {
        let real = match std::fs::canonicalize(path) {
            Ok(p) => p.to_string_lossy().into_owned(),
            Err(_) => return path.to_string(),
        };
        // Windows answers with a verbatim \\?\ path; the files woven in are
        // spelled as Go resolves them, without it, and must compare equal.
        if let Some(rest) = real.strip_prefix(r"\\?\UNC\") {
            format!(r"\\{}", rest)
        } else if let Some(rest) = real.strip_prefix(r"\\?\") {
            rest.to_string()
        } else {
            real
        }
    }

    fn is_watched(state: &State, f: &F) -> bool {
        let specs = match state.watches.iter().find(|w| w.0 == f.file) {
            Some(entry) => &entry.1,
            None => return false,
        };
        let mut named = 0;
        for spec in specs {
            if spec.symbol == "*" {
                return true;
            }
            let tail = spec.symbol.rsplit(|c| c == '.' || c == ':').next().unwrap_or("");
            if tail != f.name {
                continue;
            }
            named += 1;
            // The index's range starts at the declaration, attributes included.
            if spec.start == 0 || (spec.start <= f.line + 3 && f.line <= spec.end.max(spec.start)) {
                return true;
            }
        }
        named == 1
    }

    /// A call in progress; dropping it is the call ending.
    pub struct Call {
        f: Option<&'static F>,
        id: u32,
        start: Option<Instant>,
        /// This call's detail is wanted: the function is watched and the call is one of the sampled.
        watched: bool,
        /// Which call to the watched function this is, counting every one.
        number: u32,
        /// A value was seen on its way out, and whether it was an Err.
        seen: std::cell::Cell<bool>,
        err: std::cell::Cell<bool>,
    }

    pub fn enter(f: &'static F) -> Call {
        if !on() {
            return Call { f: None, id: 0, start: None, watched: false, number: 0, seen: std::cell::Cell::new(false), err: std::cell::Cell::new(false) };
        }
        let mut id = f.id.load(Ordering::Relaxed);
        if id & HOT != 0 {
            // Counted, and nothing else: no clock, no place in the call graph.
            let counted = LOCAL.try_with(|l| match l.try_borrow_mut() {
                Ok(mut l) => { l.count(f, id & ID, 0, 1); true }
                Err(_) => false,
            }).unwrap_or(false);
            if !counted {
                f.calls.fetch_add(1, Ordering::Relaxed);
            }
            return Call { f: None, id: 0, start: None, watched: false, number: 0, seen: std::cell::Cell::new(false), err: std::cell::Cell::new(false) };
        }
        if id == 0 {
            if let Ok(mut guard) = STATE.lock() {
                if let Some(state) = guard.as_mut() {
                    id = f.id.load(Ordering::Relaxed);
                    if id == 0 {
                        state.fns.push(f);
                        id = state.fns.len() as u32;
                        if is_watched(state, f) {
                            state.watched.insert(id, Watch::default());
                            id |= WATCHED;
                        }
                        f.id.store(id, Ordering::Relaxed);
                    }
                }
            }
        }
        let (mut watched, mut number) = (id & WATCHED != 0, 0);
        if watched {
            number = f.seen.fetch_add(1, Ordering::Relaxed) + 1;
            let mut stride = 1;
            let mut limit = FULL_CALLS;
            while number > limit && limit < u32::MAX / 2 {
                stride *= 2;
                limit *= 2;
            }
            watched = number & (stride - 1) == 0;
        }
        let id = id & ID;
        let counted = LOCAL.try_with(|l| {
            let mut l = match l.try_borrow_mut() { Ok(l) => l, Err(_) => return false };
            if let Some(caller) = l.stack.last().copied() {
                if caller != id {
                    l.edge(caller, id);
                }
            }
            l.stack.push(id);
            l.count(f, id, 0, 1);
            true
        }).unwrap_or(false);
        if !counted {
            f.calls.fetch_add(1, Ordering::Relaxed); // the thread is ending: no cache left to count in
        }
        Call { f: Some(f), id, start: Some(Instant::now()), watched, number, seen: std::cell::Cell::new(false), err: std::cell::Cell::new(false) }
    }

    /// Whether an address is on this thread's stack: a local, whose address
    /// the next call's local reuses, so it identifies nothing.
    fn on_stack(address: usize) -> bool {
        let here = 0u8;
        let here = &here as *const u8 as usize;
        (if address > here { address - here } else { here - address }) < 8 * 1024 * 1024
    }

    impl Call {
        /// Watched: its arguments and result are wanted.
        pub fn w(&self) -> bool {
            self.watched
        }

        /// The function is returning a value; `is_err` when it is an Err.
        pub fn e(&self, is_err: bool) {
            self.seen.set(true);
            self.err.set(is_err);
        }

        pub fn args(&self, values: &[String], addresses: &[usize]) {
            let f = match self.f { Some(f) => f, None => return };
            let names: Vec<&str> = if f.params.is_empty() { Vec::new() } else { f.params.split(',').collect() };
            let mut shown = String::from("{");
            let mut fingerprint = String::new();
            let mut before = Vec::new();
            let mut flat_all = Vec::new();
            let mut compound = Vec::new();
            for (i, name) in names.iter().enumerate() {
                let text = values.get(i).map(|s| s.as_str()).unwrap_or("?");
                if i > 0 { shown.push_str(", "); }
                let _ = write!(shown, "{}: {}", q(name), text);
                let mut flat = Vec::new();
                let value = parse(text);
                compound.push(!matches!(value, V::Leaf(..)));
                flatten(&value, name, 0, &mut flat);
                for (path, (_, leaf)) in &flat {
                    let _ = write!(fingerprint, "{}={}|", path, leaf);
                }
                flat_all.push(flat.clone());
                before.push((i, flat));
            }
            shown.push('}');
            let shown = clip(&shown, PREVIEW_MAX);
            let mut call = 0;
            if let Ok(mut guard) = STATE.lock() {
                if let Some(w) = guard.as_mut().and_then(|s| s.watched.get_mut(&self.id)) {
                    w.sampled += 1;
                    call = self.number;
                    for flat in &flat_all {
                        for (path, (kind, leaf)) in flat {
                            w.observe(&format!("arg.{}", path), kind, leaf);
                        }
                    }
                    for (i, name) in names.iter().enumerate() {
                        let address = addresses.get(i).copied().unwrap_or(0);
                        // A number or a string has no identity worth following
                        // (a literal's address is the same on every call).
                        if address == 0 || on_stack(address) || !compound[i] {
                            continue;
                        }
                        match w.seen.get(&address).copied() {
                            None => { w.seen.insert(address, call); }
                            Some(first) => match w.shared.iter_mut().find(|s| s.0 == *name) {
                                Some(entry) => entry.1 += 1,
                                None => w.shared.push((name.to_string(), 1, first)),
                            },
                        }
                        // The same object as on an earlier call, with different contents.
                        if *name != "self" {
                            let fields: Vec<(String, String)> = flat_all[i].iter().map(|(p, (_, t))| (p[name.len()..].trim_start_matches('.').to_string(), t.clone())).collect();
                            if let Some((earlier, was)) = w.objects.get(&address) {
                                if *earlier != call {
                                    let mut changes = Vec::new();
                                    for (k, now) in &fields {
                                        let then = was.iter().find(|e| &e.0 == k).map(|e| e.1.as_str()).unwrap_or("(absent)");
                                        if then != now && changes.len() < 4 {
                                            changes.push(format!("{{\"key\":{},\"before\":{},\"after\":{}}}", q(k), q(then), q(now)));
                                        }
                                    }
                                    if !changes.is_empty() {
                                        let example = format!("{{\"fromCall\":{},\"toCall\":{},\"changes\":[{}]}}", earlier, call, changes.join(","));
                                        match w.drift.iter().position(|d| d.0 == *name) {
                                            Some(at) => {
                                                w.drift[at].1 += 1;
                                                if w.drift[at].2.len() < 4 { w.drift[at].2.push(example); }
                                            }
                                            None if w.drift.len() < 20 => w.drift.push((name.to_string(), 1, vec![example])),
                                            None => {}
                                        }
                                    }
                                }
                            }
                            w.objects.insert(address, (call, fields));
                        }
                    }
                    if w.last_args.as_deref() == Some(fingerprint.as_str()) && !names.is_empty() {
                        w.repeats += 1;
                        if w.repeat_example.is_none() {
                            w.repeat_example = Some(format!("{{\"call\":{},\"args\":{}}}", call, q(&shown)));
                        }
                    }
                    w.last_args = Some(fingerprint);
                }
            }
            let _ = FRAME.try_with(|frames| frames.borrow_mut().push(Pending { id: self.id, call, args: shown, before, returned: None, after: Vec::new() }));
        }

        pub fn ret(&self, value: String) {
            let _ = FRAME.try_with(|frames| {
                if let Some(frame) = frames.borrow_mut().iter_mut().rev().find(|p| p.id == self.id) {
                    frame.returned = Some(value);
                }
            });
        }

        /// The reference arguments as they are once the body has run.
        pub fn after(&self, values: &[(usize, String)]) {
            let _ = FRAME.try_with(|frames| {
                if let Some(frame) = frames.borrow_mut().iter_mut().rev().find(|p| p.id == self.id) {
                    frame.after = values.to_vec();
                }
            });
        }
    }

    impl Drop for Call {
        fn drop(&mut self) {
            let f = match self.f { Some(f) => f, None => return };
            let nanos = self.start.map_or(0, |at| at.elapsed().as_nanos() as u64);
            // Nothing runs when a Rust program ends: flush when the last call
            // in progress on this thread finishes (main returning, a test
            // ending), at most five times a second.
            let flush = LOCAL.try_with(|l| {
                let mut l = match l.try_borrow_mut() { Ok(l) => l, Err(_) => return None };
                if let Some(at) = l.stack.iter().rposition(|id| *id == self.id) {
                    l.stack.truncate(at);
                }
                l.count(f, self.id, nanos, 0);
                if !l.stack.is_empty() {
                    return Some(false);
                }
                let due = f.name == "main" || l.flushed.map_or(true, |at| at.elapsed().as_millis() >= 200);
                if due {
                    l.spill();
                    l.flushed = Some(Instant::now());
                }
                Some(due)
            }).ok().flatten();
            if flush.is_none() {
                f.nanos.fetch_add(nanos, Ordering::Relaxed);
            }
            let panicking = std::thread::panicking();
            let frame = if self.watched {
                FRAME.try_with(|frames| {
                    let mut frames = frames.borrow_mut();
                    frames.iter().rposition(|p| p.id == self.id).map(|at| frames.remove(at))
                }).ok().flatten()
            } else {
                None
            };
            // How the call ended. A Result function fails by returning Err,
            // written like Go's returned errors ("error: ...").
            let mut failed: Option<String> = None;
            let mut returned: Option<String> = frame.as_ref().and_then(|p| p.returned.clone());
            // A Result function that leaves without a value passing the
            // recorder left through `?` (or a macro): with an Err.
            let propagated = !self.seen.get() && f.kind == 1 && f.captures;
            if panicking {
                failed = Some(format!("panic: {}", PANIC.try_with(|p| p.borrow().clone()).ok().flatten().unwrap_or_default()));
            } else if self.err.get() {
                let text = returned.as_deref().unwrap_or("Err(..)");
                failed = Some(format!("error: {}", clip(text.strip_prefix("Err(").and_then(|t| t.strip_suffix(')')).unwrap_or(text), 200)));
            } else if propagated {
                failed = Some("error: (propagated with `?`)".to_string());
            } else if self.watched && returned.is_none() {
                returned = Some(if f.kind == 2 && f.captures { "None".to_string() } else if f.captures { "(returned early)".to_string() } else { "(not captured)".to_string() });
            }
            if failed.is_some() {
                f.errors.fetch_add(1, Ordering::Relaxed);
            }
            if let Some(frame) = frame {
                if let Ok(mut guard) = STATE.lock() {
                    if let Some(w) = guard.as_mut().and_then(|s| s.watched.get_mut(&self.id)) {
                        let ms = nanos as f64 / 1.0e6;
                        w.total_ms += ms;
                        let mut mutated = Vec::new();
                        if let Some(what) = &failed {
                            w.errors += 1;
                            match w.exceptions.iter().position(|e| &e.0 == what) {
                                Some(at) => w.exceptions[at].1 += 1,
                                None if w.exceptions.len() < 20 => w.exceptions.push((what.clone(), 1)),
                                None => {}
                            }
                        } else if let Some(text) = &returned {
                            let value = parse(text);
                            match &value {
                                V::Leaf("number", n) => w.returns.add("number", n),
                                _ => w.returns.add("text", &clip(text, 80)),
                            }
                            let mut flat = Vec::new();
                            flatten(&value, "return", 0, &mut flat);
                            for (path, (kind, leaf)) in &flat {
                                w.observe(path, kind, leaf);
                            }
                        }
                        // What the call changed behind its references.
                        let names: Vec<&str> = if f.params.is_empty() { Vec::new() } else { f.params.split(',').collect() };
                        for (index, text) in &frame.after {
                            let name = names.get(*index).copied().unwrap_or("?");
                            let mut now = Vec::new();
                            flatten(&parse(text), name, 0, &mut now);
                            let was = frame.before.iter().find(|b| b.0 == *index).map(|b| &b.1);
                            let mut changed: Vec<(String, String, String)> = Vec::new();
                            if let Some(was) = was {
                                for (path, (_, old)) in was {
                                    let new = now.iter().find(|n| &n.0 == path).map(|n| (n.1).1.as_str()).unwrap_or("(absent)");
                                    if new != old { changed.push((path.clone(), old.clone(), new.to_string())); }
                                }
                                for (path, (_, new)) in &now {
                                    if !was.iter().any(|o| &o.0 == path) { changed.push((path.clone(), "(absent)".to_string(), new.clone())); }
                                }
                            }
                            for (path, old, new) in changed.iter().take(20) {
                                // An empty container gaining its first entry is one change, not two.
                                let placeholder = old == "{}" || old == "(absent)" || new == "{}" || new == "(absent)";
                                let child = changed.iter().any(|o| o.0 != *path && (o.0.starts_with(&format!("{}.", path)) || o.0.starts_with(&format!("{}[", path))));
                                if placeholder && child { continue; }
                                let example = format!("{{\"call\":{},\"before\":{},\"after\":{}}}", frame.call, q(old), q(new));
                                match w.mutations.iter().position(|m| &m.0 == path) {
                                    Some(at) => {
                                        w.mutations[at].1 += 1;
                                        if w.mutations[at].2.len() < 6 { w.mutations[at].2.push(example); }
                                    }
                                    None if w.mutations.len() < 40 => w.mutations.push((path.clone(), 1, vec![example])),
                                    None => {}
                                }
                                if mutated.len() < 4 { mutated.push(format!("{}: {} → {}", path, old, new)); }
                            }
                        }
                        w.sample(Sample { call: frame.call, args: frame.args, ms, returned: if failed.is_some() { None } else { returned.map(|r| clip(&r, PREVIEW_MAX)) }, threw: failed, mutated });
                    }
                }
            }
            if flush == Some(true) {
                write();
            }
        }
    }

    fn write() {
        let (doc, path) = {
            let locked = if EXITING.load(Ordering::Relaxed) {
                STATE.try_lock().map_err(|_| "held by a thread that is gone")
            } else {
                STATE.lock().map_err(|_| "poisoned")
            };
            let mut guard = match locked {
                Ok(g) => g,
                Err(why) => { debug(format_args!("could not write: the recorder's lock is {}", why)); return }
            };
            let state = match guard.as_mut() { Some(s) => s, None => return };
            state.last_flush = Instant::now();
            let total: u64 = state.fns.iter().map(|f| f.calls.load(Ordering::Relaxed) + f.errors.load(Ordering::Relaxed)).sum::<u64>() + state.uncaught.len() as u64;
            if total == state.last_written || total == 0 {
                debug(format_args!("nothing new to write ({} calls)", total));
                return;
            }
            state.last_written = total;
            let mut b = String::with_capacity(4096);
            let _ = write!(b, "{{\"version\":1,\"runId\":{},\"pid\":{},\"argv\":[\"rust\"],\"startedAt\":{},\"endedAt\":{},\"functions\":[", q(&state.run_id), std::process::id(), state.started, now_ms());
            let mut n = 0;
            for f in &state.fns {
                let calls = f.calls.load(Ordering::Relaxed);
                if calls == 0 { continue; }
                if n > 0 { b.push(','); }
                n += 1;
                let _ = write!(b, "{{\"file\":{},\"name\":{},\"line\":{},\"calls\":{},\"errors\":{},\"ms\":{}}}", q(f.file), q(f.name), f.line, calls, f.errors.load(Ordering::Relaxed), num(round3(f.nanos.load(Ordering::Relaxed) as f64 / 1.0e6)));
            }
            b.push_str("],\"edges\":[");
            let mut n = 0;
            for slot in 0..EDGE_SLOTS {
                let key = EDGE_KEYS[slot].load(Ordering::Relaxed);
                let calls = EDGE_COUNTS[slot].load(Ordering::Relaxed);
                let (from, to) = ((key >> 32) as usize, (key & 0xffff_ffff) as usize);
                if key == 0 || calls == 0 || from > state.fns.len() || to > state.fns.len() {
                    continue;
                }
                if n > 0 { b.push(','); }
                n += 1;
                let (a, z) = (state.fns[from - 1], state.fns[to - 1]);
                let _ = write!(b, "{{\"from\":{{\"file\":{},\"name\":{},\"line\":{}}},\"to\":{{\"file\":{},\"name\":{},\"line\":{}}},\"calls\":{}}}", q(a.file), q(a.name), a.line, q(z.file), q(z.name), z.line, calls);
            }
            let _ = write!(b, "],\"edgesDropped\":{},\"watched\":[", EDGES_DROPPED.load(Ordering::Relaxed));
            let mut n = 0;
            let mut ids: Vec<&u32> = state.watched.keys().collect();
            ids.sort();
            for id in ids {
                let w = &state.watched[id];
                if w.sampled == 0 { continue; }
                if n > 0 { b.push(','); }
                n += 1;
                w.json(state.fns[*id as usize - 1], &mut b);
            }
            let _ = write!(b, "],\"uncaught\":[{}]}}", state.uncaught.join(","));
            // One file per crate: each crate in the program carries its own recorder.
            (b, format!("{}/{}-{}", state.dir, std::process::id(), CRATE))
        };
        let _ = std::fs::create_dir_all(std::path::Path::new(&path).parent().unwrap_or(std::path::Path::new(".")));
        let tmp = format!("{}.json.tmp", path);
        match std::fs::write(&tmp, doc).and_then(|_| std::fs::rename(&tmp, format!("{}.json", path))) {
            Ok(()) => debug(format_args!("wrote {}.json", path)),
            Err(error) => debug(format_args!("could not write {}.json: {}", path, error)),
        }
    }

    // ── the little JSON the recorder needs ──────────────────────────────────

    fn q(s: &str) -> String {
        let mut b = String::with_capacity(s.len() + 2);
        b.push('"');
        for c in s.chars() {
            match c {
                '"' => b.push_str("\\\""),
                '\\' => b.push_str("\\\\"),
                '\n' => b.push_str("\\n"),
                '\r' => b.push_str("\\r"),
                '\t' => b.push_str("\\t"),
                c if (c as u32) < 0x20 => { let _ = write!(b, "\\u{:04x}", c as u32); }
                c => b.push(c),
            }
        }
        b.push('"');
        b
    }

    fn strings(items: &[String]) -> String {
        format!("[{}]", items.iter().map(|s| q(s)).collect::<Vec<_>>().join(","))
    }

    fn num(n: f64) -> String {
        if n.is_finite() { format!("{}", n) } else { "null".to_string() }
    }

    fn round3(n: f64) -> f64 {
        (n * 1000.0).round() / 1000.0
    }

    /// The top-level objects of a JSON array, as raw text.
    fn objects(json: &str) -> Vec<String> {
        let (mut out, mut depth, mut start, mut quoted, mut escaped) = (Vec::new(), 0, 0, false, false);
        for (i, c) in json.char_indices() {
            if quoted {
                if escaped { escaped = false; } else if c == '\\' { escaped = true; } else if c == '"' { quoted = false; }
                continue;
            }
            match c {
                '"' => quoted = true,
                '{' => {
                    if depth == 0 { start = i; }
                    depth += 1;
                }
                '}' => {
                    depth -= 1;
                    if depth == 0 { out.push(json[start..=i].to_string()); }
                }
                _ => {}
            }
        }
        out
    }

    /// A string or number field of a flat JSON object.
    fn field(object: &str, name: &str) -> Option<String> {
        let key = format!("\"{}\"", name);
        let rest = object[object.find(&key)? + key.len()..].trim_start().strip_prefix(':')?.trim_start();
        if let Some(text) = rest.strip_prefix('"') {
            let (mut out, mut escaped) = (String::new(), false);
            for c in text.chars() {
                if escaped {
                    out.push(match c { 'n' => '\n', 't' => '\t', c => c });
                    escaped = false;
                } else if c == '\\' {
                    escaped = true;
                } else if c == '"' {
                    return Some(out);
                } else {
                    out.push(c);
                }
            }
            return None;
        }
        Some(rest.chars().take_while(|c| c.is_ascii_digit() || *c == '.' || *c == '-').collect())
    }
}
